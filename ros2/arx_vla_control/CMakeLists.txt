cmake_minimum_required(VERSION 3.16)
project(arx_vla_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(arx5_arm_msg REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(nlohmann_json REQUIRED)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ARX_REPO "${PROJECT_ROOT}/drivers/arx_r5" CACHE PATH "Canonical ARX SDK checkout")
set(ARX_CONTROLLER "${ARX_REPO}/ROS2/R5_ws/src/ARX_R5_ros2_V7/arx_r5_controller")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
  set(ARX_NATIVE "${ARX_CONTROLLER}/lib/arx_r5_src/libarx_r5_src-arm64.so")
else()
  set(ARX_NATIVE "${ARX_CONTROLLER}/lib/arx_r5_src/libarx_r5_src.so")
endif()
foreach(target executor watchdog kinematics_probe core_test)
  add_executable(${target} src/${target}.cpp)
  target_include_directories(${target} PRIVATE include
    "${ARX_REPO}/py/ARX_R5_python/bimanual/lib"
    "${ARX_CONTROLLER}/lib/arx_r5_src/include"
    "${ARX_CONTROLLER}/lib/arx_hardware_interface/include")
  target_link_libraries(${target} Eigen3::Eigen nlohmann_json::nlohmann_json ${ARX_NATIVE})
  ament_target_dependencies(${target} rclcpp std_msgs std_srvs arx5_arm_msg)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(${target} PROPERTIES INSTALL_RPATH "${ARX_CONTROLLER}/lib/arx_r5_src")
  install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
endforeach()
enable_testing()
target_compile_definitions(core_test PRIVATE DEFAULT_CONFIG="${PROJECT_ROOT}/configs/real/arx_r5.json")
add_test(NAME control_core COMMAND core_test)
ament_package()
