cmake_minimum_required(VERSION 3.16)
project(pico_r5_bridge)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(arx5_arm_msg REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(nlohmann_json REQUIRED)

set(XROBO_SDK_ROOT /opt/apps/roboticsservice/SDK CACHE PATH "Installed XRoboToolkit SDK")
find_path(XROBO_INCLUDE PXREARobotSDK.h PATHS ${XROBO_SDK_ROOT}/include REQUIRED NO_DEFAULT_PATH)
find_library(XROBO_LIBRARY PXREARobotSDK PATHS ${XROBO_SDK_ROOT}/arm64 ${XROBO_SDK_ROOT}/linux/64 REQUIRED NO_DEFAULT_PATH)
get_filename_component(XROBO_LIBRARY_DIR ${XROBO_LIBRARY} DIRECTORY)
add_executable(pico_preview src/pico_preview.cpp)
target_include_directories(pico_preview PRIVATE include ${XROBO_INCLUDE})
target_link_libraries(pico_preview Eigen3::Eigen nlohmann_json::nlohmann_json ${XROBO_LIBRARY})
ament_target_dependencies(pico_preview rclcpp geometry_msgs std_msgs)
set_target_properties(pico_preview PROPERTIES INSTALL_RPATH "${XROBO_LIBRARY_DIR}")
target_compile_options(pico_preview PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS pico_preview DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})

# Reuse the installed project's native solver; the separate Python solver .so is x86-only.
get_filename_component(R5_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(ARX_CONTROLLER "${R5_REPO}/ROS2/R5_ws/src/ARX_R5_ros2_V7/arx_r5_controller")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
  set(ARX_NATIVE "${ARX_CONTROLLER}/lib/arx_r5_src/libarx_r5_src-arm64.so")
else()
  set(ARX_NATIVE "${ARX_CONTROLLER}/lib/arx_r5_src/libarx_r5_src.so")
endif()
set(ARX_INCLUDES "${R5_REPO}/py/ARX_R5_python/bimanual/lib"
  "${ARX_CONTROLLER}/lib/arx_r5_src/include"
  "${ARX_CONTROLLER}/lib/arx_hardware_interface/include")
add_executable(pico_ik_bridge src/pico_ik_bridge.cpp)
target_include_directories(pico_ik_bridge PRIVATE include ${ARX_INCLUDES})
target_link_libraries(pico_ik_bridge Eigen3::Eigen nlohmann_json::nlohmann_json ${ARX_NATIVE})
ament_target_dependencies(pico_ik_bridge rclcpp std_msgs arx5_arm_msg)
target_compile_options(pico_ik_bridge PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(pico_ik_bridge PROPERTIES INSTALL_RPATH "${ARX_CONTROLLER}/lib/arx_r5_src")
install(TARGETS pico_ik_bridge DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  enable_testing()
  add_executable(preview_test tests/preview_test.cpp)
  target_include_directories(preview_test PRIVATE include)
  target_link_libraries(preview_test Eigen3::Eigen nlohmann_json::nlohmann_json)
  add_test(NAME preview_safety COMMAND preview_test)
  add_executable(joystick_test tests/joystick_test.cpp)
  target_include_directories(joystick_test PRIVATE include)
  target_link_libraries(joystick_test Eigen3::Eigen nlohmann_json::nlohmann_json)
  add_test(NAME joystick_mapping COMMAND joystick_test)
  add_executable(arx_ik_probe tests/arx_ik_probe.cpp)
  target_include_directories(arx_ik_probe PRIVATE ${ARX_INCLUDES})
  target_link_libraries(arx_ik_probe Eigen3::Eigen ${ARX_NATIVE})
  add_test(NAME arx_ik_roundtrip COMMAND arx_ik_probe)
endif()
ament_package()
